cmake_minimum_required(VERSION 3.16)
project(FakeMobileGames VERSION 1.14.0 LANGUAGES CXX)
include_directories(
  "${CMAKE_CURRENT_LIST_DIR}/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/juices-sort/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/new-blocks/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/color-mix/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/sudoku/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/traffic-jam/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/mahjong/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/boca/src"
  "${CMAKE_CURRENT_LIST_DIR}/../../games/color-loop/src"
)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FMG_CORE_ONLY "Build rule tests without SDL2" OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
if(NOT FMG_CORE_ONLY)
 find_package(SDL2 REQUIRED CONFIG)
 add_executable(fake-mobile-games src/main.cpp)
 target_link_libraries(fake-mobile-games PRIVATE SDL2::SDL2)
 install(TARGETS fake-mobile-games RUNTIME DESTINATION bin)
 install(FILES ${CMAKE_CURRENT_LIST_DIR}/../../games/boca/assets/boca-atlas-v2.bmp ${CMAKE_CURRENT_LIST_DIR}/../../games/boca/assets/boca-ground.bmp DESTINATION assets)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(fmg-tests tests/tests.cpp)
 target_include_directories(fmg-tests PRIVATE src)
 add_test(NAME rules-and-save COMMAND fmg-tests)
endif()

if(BUILD_TESTING AND NOT FMG_CORE_ONLY AND UNIX)
 find_package(Threads REQUIRED)
 add_executable(fmg-ui-tests tests/juice_ui.cpp)
 target_link_libraries(fmg-ui-tests PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME juice-controls-and-resume COMMAND fmg-ui-tests)
 set_tests_properties(juice-controls-and-resume PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
endif()
