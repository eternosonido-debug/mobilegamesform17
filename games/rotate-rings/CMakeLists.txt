cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(RotateRings VERSION 0.2.0 LANGUAGES CXX)
 include(CTest)
endif()
find_package(SDL2 REQUIRED CONFIG)
add_executable(rotate-rings src/main.cpp)
target_compile_features(rotate-rings PRIVATE cxx_std_11)
target_include_directories(rotate-rings PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(rotate-rings PRIVATE SDL2::SDL2)
install(TARGETS rotate-rings RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(rotate-rings-tests tests/rules.cpp)
 target_compile_features(rotate-rings-tests PRIVATE cxx_std_11)
 target_include_directories(rotate-rings-tests PRIVATE src)
 add_test(NAME rotate-rings-rules-save-solutions COMMAND rotate-rings-tests)
 add_test(NAME rotate-rings-sdl-smoke COMMAND rotate-rings --smoke)
 set_tests_properties(rotate-rings-sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(rotate-rings-controls tests/controls.cpp)
 target_compile_features(rotate-rings-controls PRIVATE cxx_std_11)
 target_include_directories(rotate-rings-controls PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(rotate-rings-controls PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME rotate-rings-controls COMMAND rotate-rings-controls)
 set_tests_properties(rotate-rings-controls PROPERTIES TIMEOUT 15 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(rotate-rings-workflow tests/workflow.cpp)
 target_compile_features(rotate-rings-workflow PRIVATE cxx_std_11)
 target_include_directories(rotate-rings-workflow PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(rotate-rings-workflow PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME rotate-rings-victory-unlock COMMAND rotate-rings-workflow)
 set_tests_properties(rotate-rings-victory-unlock PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
