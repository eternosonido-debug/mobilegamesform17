cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(TowerBreak VERSION 0.1.0 LANGUAGES CXX)
 include(CTest)
endif()
find_package(SDL2 REQUIRED CONFIG)
add_executable(tower-break src/main.cpp)
target_compile_features(tower-break PRIVATE cxx_std_11)
target_include_directories(tower-break PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(tower-break PRIVATE SDL2::SDL2)
install(TARGETS tower-break RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(tower-break-tests tests/rules.cpp)
 target_compile_features(tower-break-tests PRIVATE cxx_std_11)
 target_include_directories(tower-break-tests PRIVATE src)
 add_test(NAME tower-break-rules-save-solutions COMMAND tower-break-tests)
 add_test(NAME tower-break-sdl-smoke COMMAND tower-break --smoke)
 set_tests_properties(tower-break-sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(tower-break-controls tests/controls.cpp)
 target_compile_features(tower-break-controls PRIVATE cxx_std_11)
 target_include_directories(tower-break-controls PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(tower-break-controls PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME tower-break-controls COMMAND tower-break-controls)
 set_tests_properties(tower-break-controls PROPERTIES TIMEOUT 15 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(tower-break-workflow tests/workflow.cpp)
 target_compile_features(tower-break-workflow PRIVATE cxx_std_11)
 target_include_directories(tower-break-workflow PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(tower-break-workflow PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME tower-break-victory-unlock COMMAND tower-break-workflow)
 set_tests_properties(tower-break-victory-unlock PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
