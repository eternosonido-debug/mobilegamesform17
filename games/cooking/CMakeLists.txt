cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(Cooking VERSION 0.1.0 LANGUAGES CXX)
 include(CTest)
endif()
find_package(SDL2 REQUIRED CONFIG)
add_executable(cooking src/main.cpp)
target_compile_features(cooking PRIVATE cxx_std_11)
target_include_directories(cooking PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(cooking PRIVATE SDL2::SDL2)
target_compile_definitions(cooking PRIVATE COOKING_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
install(DIRECTORY assets/ DESTINATION share/cooking)
install(TARGETS cooking RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(cooking-tests tests/rules.cpp)
 target_compile_features(cooking-tests PRIVATE cxx_std_11)
 target_include_directories(cooking-tests PRIVATE src)
 add_test(NAME cooking-recipes-save COMMAND cooking-tests)
 set_tests_properties(cooking-recipes-save PROPERTIES TIMEOUT 20)
 add_test(NAME cooking-sdl-smoke COMMAND cooking --smoke)
 set_tests_properties(cooking-sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
 find_package(Threads REQUIRED)
 add_executable(cooking-controls tests/controls.cpp)
 target_compile_features(cooking-controls PRIVATE cxx_std_11)
 target_include_directories(cooking-controls PRIVATE ../../shared/fmg-runtime/src)
 target_compile_definitions(cooking-controls PRIVATE COOKING_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
 target_link_libraries(cooking-controls PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME cooking-controls COMMAND cooking-controls)
 set_tests_properties(cooking-controls PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(cooking-victory tests/victory.cpp)
 target_compile_features(cooking-victory PRIVATE cxx_std_11)
 target_include_directories(cooking-victory PRIVATE ../../shared/fmg-runtime/src)
 target_compile_definitions(cooking-victory PRIVATE COOKING_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")

endif()
