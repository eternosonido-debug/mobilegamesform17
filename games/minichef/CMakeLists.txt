cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(Minichef VERSION 0.1.0 LANGUAGES CXX)
 include(CTest)
endif()
find_package(SDL2 REQUIRED CONFIG)
add_executable(minichef src/main.cpp)
target_compile_features(minichef PRIVATE cxx_std_11)
target_include_directories(minichef PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(minichef PRIVATE SDL2::SDL2)
install(TARGETS minichef RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(minichef-tests tests/rules.cpp)
 target_compile_features(minichef-tests PRIVATE cxx_std_11)
 target_include_directories(minichef-tests PRIVATE src)
 add_test(NAME minichef-recipes-save COMMAND minichef-tests)
 set_tests_properties(minichef-recipes-save PROPERTIES TIMEOUT 20)
 add_test(NAME minichef-sdl-smoke COMMAND minichef --smoke)
 set_tests_properties(minichef-sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
 find_package(Threads REQUIRED)
 add_executable(minichef-controls tests/controls.cpp)
 target_compile_features(minichef-controls PRIVATE cxx_std_11)
 target_include_directories(minichef-controls PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(minichef-controls PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME minichef-controls COMMAND minichef-controls)
 set_tests_properties(minichef-controls PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(minichef-victory tests/victory.cpp)
 target_compile_features(minichef-victory PRIVATE cxx_std_11)
 target_include_directories(minichef-victory PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(minichef-victory PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME minichef-victory-unlock COMMAND minichef-victory)
 set_tests_properties(minichef-victory-unlock PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
