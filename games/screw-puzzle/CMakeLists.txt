cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ScrewPuzzle VERSION 0.1.0 LANGUAGES CXX)
 include(CTest)
endif()
find_package(SDL2 REQUIRED CONFIG)
add_executable(screw-puzzle src/main.cpp)
target_compile_features(screw-puzzle PRIVATE cxx_std_11)
target_include_directories(screw-puzzle PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(screw-puzzle PRIVATE SDL2::SDL2)
install(TARGETS screw-puzzle RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(screw-puzzle-tests tests/rules.cpp)
 target_compile_features(screw-puzzle-tests PRIVATE cxx_std_11)
 target_include_directories(screw-puzzle-tests PRIVATE src)
 add_test(NAME screw-puzzle-rules-save-solutions COMMAND screw-puzzle-tests)
 add_test(NAME screw-puzzle-sdl-smoke COMMAND screw-puzzle --smoke)
 set_tests_properties(screw-puzzle-sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(screw-puzzle-controls tests/controls.cpp)
 target_compile_features(screw-puzzle-controls PRIVATE cxx_std_11)
 target_include_directories(screw-puzzle-controls PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(screw-puzzle-controls PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME screw-puzzle-controls COMMAND screw-puzzle-controls)
 set_tests_properties(screw-puzzle-controls PROPERTIES TIMEOUT 15 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
