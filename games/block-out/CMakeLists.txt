cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(BlockOut VERSION 0.1.0 LANGUAGES CXX)
 include(CTest)
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL2 REQUIRED CONFIG)
add_executable(block-out src/main.cpp)
target_compile_features(block-out PRIVATE cxx_std_11)
target_include_directories(block-out PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(block-out PRIVATE SDL2::SDL2)
install(TARGETS block-out RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(block-out-tests tests/rules.cpp)
 target_compile_features(block-out-tests PRIVATE cxx_std_11)
 target_include_directories(block-out-tests PRIVATE src)
 add_test(NAME block-out-rules-save COMMAND block-out-tests)
 find_package(Threads REQUIRED)
 add_executable(block-out-ui-tests tests/controls.cpp)
 target_compile_features(block-out-ui-tests PRIVATE cxx_std_11)
 target_include_directories(block-out-ui-tests PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(block-out-ui-tests PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME block-out-controls COMMAND block-out-ui-tests)
 set_tests_properties(block-out-controls PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(block-out-visual-tests tests/exit_visual.cpp)
 target_compile_features(block-out-visual-tests PRIVATE cxx_std_11)
 target_include_directories(block-out-visual-tests PRIVATE src)
 target_link_libraries(block-out-visual-tests PRIVATE SDL2::SDL2)
 add_test(NAME block-out-exit-boundary COMMAND block-out-visual-tests)
 set_tests_properties(block-out-exit-boundary PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
endif()
