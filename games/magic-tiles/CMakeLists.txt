cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(MagicTiles VERSION 0.2.0 LANGUAGES CXX)
 include(CTest)
endif()
find_package(SDL2 REQUIRED CONFIG)
add_executable(magic-tiles src/main.cpp)
target_compile_features(magic-tiles PRIVATE cxx_std_11)
target_include_directories(magic-tiles PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(magic-tiles PRIVATE SDL2::SDL2)
install(TARGETS magic-tiles RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(magic-tiles-tests tests/rules.cpp)
 target_compile_features(magic-tiles-tests PRIVATE cxx_std_11)
 target_include_directories(magic-tiles-tests PRIVATE src)
 add_test(NAME magic-tiles-rules-save-solutions COMMAND magic-tiles-tests)
 add_test(NAME magic-tiles-sdl-smoke COMMAND magic-tiles --smoke)
 set_tests_properties(magic-tiles-sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(magic-tiles-controls tests/controls.cpp)
 target_compile_features(magic-tiles-controls PRIVATE cxx_std_11)
 target_include_directories(magic-tiles-controls PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(magic-tiles-controls PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME magic-tiles-controls COMMAND magic-tiles-controls)
 set_tests_properties(magic-tiles-controls PROPERTIES TIMEOUT 15 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(magic-tiles-audio tests/audio.cpp)
 target_compile_features(magic-tiles-audio PRIVATE cxx_std_11)
 target_include_directories(magic-tiles-audio PRIVATE src)
 target_link_libraries(magic-tiles-audio PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME magic-tiles-audio-clock COMMAND magic-tiles-audio)
 set_tests_properties(magic-tiles-audio-clock PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_AUDIODRIVER=dummy")
endif()
