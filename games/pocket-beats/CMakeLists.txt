cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(PocketBeats VERSION 0.6.0 LANGUAGES CXX)
 include(CTest)
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL2 REQUIRED CONFIG)
add_executable(pocket-beats src/main.cpp)
target_include_directories(pocket-beats PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(pocket-beats PRIVATE SDL2::SDL2)
install(TARGETS pocket-beats RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(pocket-beats-rules tests/rules.cpp)
 add_test(NAME pocket-beats-rules-save COMMAND pocket-beats-rules)
 add_executable(pocket-beats-audio tests/audio.cpp)
 add_test(NAME pocket-beats-four-instruments COMMAND pocket-beats-audio)
 add_executable(pocket-beats-controls tests/controls.cpp)
 target_include_directories(pocket-beats-controls PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(pocket-beats-controls PRIVATE SDL2::SDL2)
 add_test(NAME pocket-beats-controls COMMAND pocket-beats-controls)
 set_tests_properties(pocket-beats-controls PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(pocket-beats-realtime tests/realtime.cpp)
 target_link_libraries(pocket-beats-realtime PRIVATE SDL2::SDL2)
 add_test(NAME pocket-beats-audio-with-ui-stalled COMMAND pocket-beats-realtime)
 set_tests_properties(pocket-beats-audio-with-ui-stalled PROPERTIES TIMEOUT 15 ENVIRONMENT "SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(pocket-beats-workflow tests/workflow.cpp)
 target_include_directories(pocket-beats-workflow PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(pocket-beats-workflow PRIVATE SDL2::SDL2)
 add_test(NAME pocket-beats-direct-notes-save-variation COMMAND pocket-beats-workflow)
endif()
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(pocket-beats-ui-flow tests/ui_flow.cpp)
 target_include_directories(pocket-beats-ui-flow PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(pocket-beats-ui-flow PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME pocket-beats-window-notes-play-save COMMAND pocket-beats-ui-flow)
 set_tests_properties(pocket-beats-window-notes-play-save PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(pocket-beats-full-grid tests/full_grid.cpp)
 target_include_directories(pocket-beats-full-grid PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(pocket-beats-full-grid PRIVATE SDL2::SDL2)
 add_test(NAME pocket-beats-all-steps-one-screen COMMAND pocket-beats-full-grid)
 set_tests_properties(pocket-beats-all-steps-one-screen PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
endif()
