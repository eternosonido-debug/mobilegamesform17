cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ColorPop VERSION 0.1.0 LANGUAGES CXX)
 include(CTest)
endif()
find_package(SDL2 REQUIRED CONFIG)
add_executable(color-pop src/main.cpp)
target_compile_features(color-pop PRIVATE cxx_std_11)
target_include_directories(color-pop PRIVATE ../../shared/fmg-runtime/src)
target_link_libraries(color-pop PRIVATE SDL2::SDL2)
install(TARGETS color-pop RUNTIME DESTINATION bin)
if(BUILD_TESTING)
 add_executable(color-pop-tests tests/rules.cpp)
 target_compile_features(color-pop-tests PRIVATE cxx_std_11)
 target_include_directories(color-pop-tests PRIVATE src)
 add_test(NAME color-pop-rules-save-solutions COMMAND color-pop-tests)
 add_test(NAME color-pop-sdl-smoke COMMAND color-pop --smoke)
 set_tests_properties(color-pop-sdl-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(color-pop-controls tests/controls.cpp)
 target_compile_features(color-pop-controls PRIVATE cxx_std_11)
 target_include_directories(color-pop-controls PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(color-pop-controls PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME color-pop-controls COMMAND color-pop-controls)
 set_tests_properties(color-pop-controls PROPERTIES TIMEOUT 15 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
if(BUILD_TESTING)
 add_executable(color-pop-workflow tests/workflow.cpp)
 target_compile_features(color-pop-workflow PRIVATE cxx_std_11)
 target_include_directories(color-pop-workflow PRIVATE ../../shared/fmg-runtime/src)
 target_link_libraries(color-pop-workflow PRIVATE SDL2::SDL2 Threads::Threads)
 add_test(NAME color-pop-victory-unlock COMMAND color-pop-workflow)
 set_tests_properties(color-pop-victory-unlock PROPERTIES TIMEOUT 20 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software;SDL_AUDIODRIVER=dummy")
endif()
